    local function CanSeeTarget(targetPos)
        local origin = Camera.CFrame.Position
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {LocalPlayer.Character}
        local result = Workspace:Raycast(origin, (targetPos - origin), params)
        if result then
            local hitDist = (result.Position - origin).Magnitude
            local targetDist = (targetPos - origin).Magnitude
            return hitDist >= targetDist
        end
        return true
    end

    local function GetVelocity(part)

    local function CollectTargets()
        local list = {}
        local ViewportCenter = Camera.ViewportSize / 2

        for _, Player in ipairs(Players:GetPlayers()) do
            if (Player == LocalPlayer) then continue end

            if (TeamCheckToggle.Value and IsTeammate(Player)) then continue end

            local Character = GetCharacter(Player)
            if (not IsAlive(Character)) then continue end

            if (KnockedCheckToggle.Value and IsKnocked(Character)) then continue end

            local RootPart = Character:FindFirstChild("HumanoidRootPart")
            if (not RootPart) then continue end

            local Range = (RootPart.Position - Camera.CFrame.Position).Magnitude
            if (Range > MaxDistanceSlider.Value) then continue end

            local Part = Character:FindFirstChild(HitboxDropdown.Value) or RootPart

            local ScreenPos, OnScreen = Camera:WorldToScreenPoint(Part.Position)
            if (not OnScreen) then continue end

            local ScreenDistance = (Vector2.new(ScreenPos.X, ScreenPos.Y) - ViewportCenter).Magnitude
            if (ScreenDistance > FOVSlider.Value) then continue end

            if (WallCheckToggle.Value and not CanSeeTarget(Part.Position)) then continue end

            local health = 999
            local Hum = Character:FindFirstChildOfClass("Humanoid")
            if Hum then health = Hum.Health end

            table.insert(list, {
                Player = Player,
                Part = Part,
                Root = RootPart,
                ScreenDist = ScreenDistance,
                WorldDist = Range,
                Health = health,
            })
        end

        return list
    end

    local function RankTargets(list)
        local mode = PriorityDropdown.Value
        if mode == "Crosshair" then
            table.sort(list, function(a, b) return a.ScreenDist < b.ScreenDist end)
        elseif mode == "Distance" then
            table.sort(list, function(a, b) return a.WorldDist < b.WorldDist end)
        elseif mode == "Health" then
            table.sort(list, function(a, b) return a.Health < b.Health end)
        elseif mode == "FOV + Distance" then
            table.sort(list, function(a, b)
                return (a.ScreenDist + a.WorldDist * 0.5) < (b.ScreenDist + b.WorldDist * 0.5)
            end)
        end
        return list[1]
    end

    local function GetConfigList()
        local out = {}
        for _, file in ipairs(listfiles(CONFIG_FOLDER)) do
            local Name = file:match("([^/\\]+)%.json$")
            if (Name) then
                table.insert(out, Name)
            end
        end
        return out
    end

    local function RefreshConfigList()
        ConfigListDropdown.Values = GetConfigList()
        ConfigListDropdown:SetValues()
        ConfigListDropdown:SetValue(nil)
    end

    local function SaveConfig(Name)
        if (not Name or Name:gsub("%s", "") == "") then
            Library:Notify("Invalid config name (empty)", 2)
            return
        end

        local Data = { objects = {} }

        for Idx, Toggle in next, SaveToggles do
            table.insert(Data.objects, { type = "Toggle", idx = Idx, value = Toggle.Value })
        end

        for Idx, Option in next, SaveOptions do
            local Type = Option.Type
            if (Type == "Slider") then
                table.insert(Data.objects, { type = "Slider", idx = Idx, value = tostring(Option.Value) })
            elseif (Type == "Dropdown") then
                table.insert(Data.objects, { type = "Dropdown", idx = Idx, value = Option.Value, mutli = Option.Multi })
            elseif (Type == "KeyPicker") then
                table.insert(Data.objects, { type = "KeyPicker", idx = Idx, mode = Option.Mode, key = Option.Value })
            end
        end

        local Success, Encoded = pcall(HttpService.JSONEncode, HttpService, Data)
        if (not Success) then
            Library:Notify("Failed to save config: encode error", 3)
            return
        end

        writefile(CONFIG_FOLDER .. "/" .. Name .. ".json", Encoded)
        Library:Notify(string.format('Created config "%s"', Name), 3)
        RefreshConfigList()
    end

    local function LoadConfig(Name)
        if (not Name) then
            Library:Notify("No config selected", 2)
            return
        end

        local Path = CONFIG_FOLDER .. "/" .. Name .. ".json"
        if (not isfile(Path)) then
            Library:Notify("Config not found", 2)
            return
        end

        local Success, Decoded = pcall(HttpService.JSONDecode, HttpService, readfile(Path))
        if (not Success) then
            Library:Notify("Failed to load config: decode error", 3)
            return
        end

        for _, Object in next, Decoded.objects do
            local Toggle = SaveToggles[Object.idx]
            local Option = SaveOptions[Object.idx]

            if (Toggle and Object.type == "Toggle") then
                Toggle:SetValue(Object.value)
            elseif (Option) then
                if (Object.type == "Slider") then
                    Option:SetValue(Object.value)
                elseif (Object.type == "Dropdown") then
                    Option:SetValue(Object.value)
                elseif (Object.type == "KeyPicker") then
                    Option:SetValue({ Object.key, Object.mode })
                end
            end
        end

        Library:Notify(string.format('Loaded config "%s"', Name), 3)
    end
