    label.Font = Enum.Font.SourceSansSemibold
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = sliderContainer

    sliderBg.Position = UDim2.new(0, 0, 0, 30)
    sliderBg.BackgroundColor3 = currentButtonColor
    sliderBg.BorderSizePixel = 0

    local sCorner = Instance.new("UICorner")
    sCorner.CornerRadius = UDim.new(1, 0)
    sCorner.Parent = sliderBg

    local sliderFill = Instance.new("Frame")
    sliderFill.Size = UDim2.new((defaultVal - minVal) / (maxVal - minVal), 0, 1, 0)
    sliderFill.BackgroundColor3 = currentAccentColor
    sliderFill.BorderSizePixel = 0
    sliderFill.Parent = sliderBg
    table.insert(themeElements, {Type = "AccentFill", Instance = sliderFill})

    local fCorner = Instance.new("UICorner")
    fCorner.CornerRadius = UDim.new(1, 0)
    fCorner.Parent = sliderFill

    local draggingSlider = false

    local function updateInput(input)
        local pos = math.clamp((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
        local val = math.floor(minVal + ((maxVal - minVal) * pos))
        sliderFill.Size = UDim2.new(pos, 0, 1, 0)
        label.Text = titleText .. ": " .. tostring(val)
        if callback then
            callback(val)
        end
    end

    sliderBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            draggingSlider = true
            updateInput(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            draggingSlider = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if draggingSlider and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateInput(input)
        end
    end)
end

local function isKeybindHeld(key)
    if typeof(key) == "EnumItem" then
        if key.EnumType == Enum.KeyCode then
            return UserInputService:IsKeyDown(key)
        elseif key.EnumType == Enum.UserInputType then
            return UserInputService:IsMouseButtonPressed(key)
        end
    end
    return false
end

                local char = p.Character

    local shortestDistance = math.huge
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild(targetPartName) then
            local targetChar = p.Character
            local humanoid = targetChar:FindFirstChildOfClass("Humanoid")

                local partPos = targetChar[targetPartName].Position
                local screenPoint, onScreen = camera:WorldToViewportPoint(partPos)
                
                if onScreen then
                    local targetScreenPos = Vector2.new(screenPoint.X, screenPoint.Y)
                    local distance = (targetScreenPos - mousePos).Magnitude
                    
                    if distance < shortestDistance then
                        shortestDistance = distance
                        closestTarget = targetChar[targetPartName]
                    end
                end
            end
        end
    end
    
    if closestTarget then
        local currentCFrame = camera.CFrame
        local targetCFrame = CFrame.new(currentCFrame.Position, closestTarget.Position)
        local interpolationFactor = math.clamp(aimbotStrength * 0.1, 0.05, 1)
        camera.CFrame = currentCFrame:Lerp(targetCFrame, interpolationFactor)
    end

local espActive = false
local activeHighlights = {}
local activeBillboards = {}

local function updateESP(p)
    if not p.Character then return end
    local char = p.Character
    
    if activeHighlights[char] then
        activeHighlights[char]:Destroy()
        activeHighlights[char] = nil
    end
    if activeBillboards[char] then
        activeBillboards[char]:Destroy()
        activeBillboards[char] = nil
    end

    if espActive and p ~= player then
        local highlight = Instance.new("Highlight")
        highlight.Name = "VisualESP"
        highlight.Adornee = char
        highlight.FillColor = currentAccentColor
        highlight.FillTransparency = 0.5
        highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
        highlight.OutlineTransparency = 0
        highlight.Parent = char
        activeHighlights[char] = highlight
        table.insert(themeElements, {Type = "ESPHighlight", Instance = highlight})

        local head = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
        if head then
            local billboard = Instance.new("BillboardGui")
            billboard.Name = "ESPInfo"
            billboard.Adornee = head
            billboard.Size = UDim2.new(0, 150, 0, 40)
            billboard.StudsOffset = Vector3.new(0, 2.5, 0)
            billboard.AlwaysOnTop = true
            billboard.Parent = char

            local textLabel = Instance.new("TextLabel")
            textLabel.Name = "InfoText"
            textLabel.Size = UDim2.new(1, 0, 1, 0)
            textLabel.BackgroundTransparency = 1
            textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
            textLabel.TextSize = 13
            textLabel.Font = Enum.Font.SourceSansBold
            textLabel.TextStrokeTransparency = 0.5
            textLabel.Parent = billboard

            local connection
            connection = RunService.RenderStepped:Connect(function()
                if not char or not char.Parent or not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
                    if connection then connection:Disconnect() end
                    return
                end
                local myRoot = player.Character.HumanoidRootPart
                local targetRoot = char:FindFirstChild("HumanoidRootPart")
                if targetRoot then
                    local dist = math.floor((myRoot.Position - targetRoot.Position).Magnitude)
                    textLabel.Text = p.Name .. " [" .. tostring(dist) .. " studs]"
                end
            end)

            activeBillboards[char] = billboard
        end
    end
end

task.spawn(function()
    while true do
        pcall(function()
            if hudFrame.Visible then
                local DISPLAYENTRIES = {}
                for _, instance in ipairs(Players:GetChildren()) do
                    if not instance:IsA("Player") then continue end
                    local pObj = instance
                    local pName = pObj.Name
                    local SETITEMS = {}
                    local ITEMTODISPLAY = {}
                    local ITEMCHECK = {}

                    local backpack = pObj:FindFirstChild("Backpack")
                    if backpack then
                        for _, tool in ipairs(backpack:GetChildren()) do
                            if tool:IsA("Tool") then table.insert(ITEMCHECK, tool) end
                        end
                    end

                    local character = pObj.Character
                    if character then
                        for _, child in ipairs(character:GetChildren()) do
                            if child:IsA("Tool") then table.insert(ITEMCHECK, child) end
                        end
                    end

                    for _, tool in ipairs(ITEMCHECK) do
                        local identified = false
                        for _, part in ipairs(tool:GetChildren()) do
                            local partName = part.Name
                            local prefix = "Meshes/"
                            if string.sub(partName, 1, #prefix) == prefix then
                                local subName = string.lower(string.sub(partName, #prefix + 1))
                                for _, entry in ipairs(ITEM_DATABASE) do
                                    local pattern = string.lower(entry.meshPattern)
                                    local matched = false
                                    if entry.isPrefix then
                                        if string.sub(subName, 1, #pattern) == pattern then matched = true end
                                    else
                                        if subName == pattern then matched = true end
                                    end

                                    if matched then
                                        if not SETITEMS[entry.displayName] then
                                            SETITEMS[entry.displayName] = true
                                            table.insert(ITEMTODISPLAY, "- " .. entry.displayName)
                                        end
                                        identified = true
                                        break
                                    end
                                end
                            elseif partName == "Butt" then
                                if not SETITEMS["Axe"] then
                                    SETITEMS["Axe"] = true
                                    table.insert(ITEMTODISPLAY, "- Axe")
                                end
                                identified = true
                            elseif partName == "Bat" then
                                if not SETITEMS["Bat"] then
                                    SETITEMS["Bat"] = true
                                    table.insert(ITEMTODISPLAY, "- Bat")
                                end
                                identified = true
                            end
                            if identified then break end
                        end
                    end

                    if #ITEMTODISPLAY > 0 then
                        table.insert(DISPLAYENTRIES, pName .. ":\n" .. table.concat(ITEMTODISPLAY, "\n"))
                    end
                end

                local resultStr = table.concat(DISPLAYENTRIES, "\n\n")
                if resultStr == "" then
                    hudTextLabel.Text = "No player inventories detected."
                else
                    hudTextLabel.Text = resultStr
                end
            end
        end)
        task.wait(0.5)
    end
end)

    label.Parent = container

    local button = Instance.new("TextButton")
    button.Size = UDim2.new(0,80,0,24)
    button.Position = UDim2.new(1,-80,0.5,-12)
    button.BackgroundColor3 = currentButtonColor
    button.Text = uiToggleKey.Name
    button.TextColor3 = Color3.fromRGB(212,212,216)
    button.TextSize = 12
    button.Font = Enum.Font.SourceSansBold
    button.Parent = container

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0,4)
    corner.Parent = button

    local waiting = false
    button.MouseButton1Click:Connect(function()
        if waiting then return end
        waiting = true
        button.Text = "Press..."
        local con
        con = UserInputService.InputBegan:Connect(function(input, gp)
            if gp then return end
            if input.UserInputType == Enum.UserInputType.Keyboard then
                uiToggleKey = input.KeyCode
                button.Text = input.KeyCode.Name
                waiting = false
                con:Disconnect()
            end
        end)
    end)
end

createDropdown(uiSettingsTab, "Color Theme Style", {"Midnight Blue", "Dark", "Deep Ocean", "Slate"}, "Midnight Blue", function(selected)
    for _, item in ipairs(themeElements) do
        if item.Type == "Background" then
            if selected == "Midnight Blue" then
                currentBgColor = Color3.fromRGB(6, 9, 18)
                currentButtonColor = Color3.fromRGB(12, 16, 28)
            elseif selected == "Dark" then
                currentBgColor = Color3.fromRGB(10, 10, 12)
                currentButtonColor = Color3.fromRGB(18, 18, 22)
            elseif selected == "Deep Ocean" then
                currentBgColor = Color3.fromRGB(4, 18, 35)
                currentButtonColor = Color3.fromRGB(10, 28, 50)
            elseif selected == "Slate" then
                currentBgColor = Color3.fromRGB(18, 21, 28)
                currentButtonColor = Color3.fromRGB(26, 31, 40)
            end
            
            -- Apply background change
            item.Instance.BackgroundColor3 = currentBgColor
        end
    end

    -- Automatically sync button/element colors to match the selected theme background nicely
    for _, item in ipairs(themeElements) do
        if item.Type == "ButtonBg" then
            item.Instance.BackgroundColor3 = currentButtonColor
        end
    end
end)

createDropdown(uiSettingsTab, "Accent Color", {"Indigo", "Emerald", "Ruby", "Gold", "Cyan"}, "Indigo", function(selected)
    if selected == "Indigo" then
        currentAccentColor = Color3.fromRGB(99, 102, 241)
    elseif selected == "Emerald" then
        currentAccentColor = Color3.fromRGB(16, 185, 129)
    elseif selected == "Ruby" then
        currentAccentColor = Color3.fromRGB(239, 68, 68)
    elseif selected == "Gold" then
        currentAccentColor = Color3.fromRGB(245, 158, 11)
    elseif selected == "Cyan" then
        currentAccentColor = Color3.fromRGB(6, 182, 212)
    end

    -- Update active accent elements dynamically
    for _, item in ipairs(themeElements) do
        if item.Type == "AccentFill" or item.Type == "AccentButton" then
            item.Instance.BackgroundColor3 = currentAccentColor
        elseif item.Type == "ESPHighlight" and item.Instance.Parent then
            item.Instance.FillColor = currentAccentColor
        end
    end
end)
