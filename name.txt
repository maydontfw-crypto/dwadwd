-- // Helpers
local function GetCharacter(Player)
    return Player.Character
end

local function IsAlive(Character)
    if (not Character) then
        return false
    end

    local Humanoid = Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Character:FindFirstChild("HumanoidRootPart")
    return Humanoid and RootPart and Humanoid.Health > 0
end

-- // Find the best target and return the aim point (or nil)
local function GetTargetPoint()
    local ViewportCenter = Camera.ViewportSize / 2
    local BestPart = nil
    local BestDistance = math.huge

    for _, Player in ipairs(Players:GetPlayers()) do
        if (Player == LocalPlayer) then
            continue
        end

        if (TeamCheckToggle.Value and LocalPlayer.Team and Player.Team and Player.Team == LocalPlayer.Team) then
            continue
        end

        local Character = GetCharacter(Player)
        if (not IsAlive(Character)) then
            continue
        end

        local RootPart = Character:FindFirstChild("HumanoidRootPart")
        if (not RootPart) then
            continue
        end

        local Range = (RootPart.Position - Camera.CFrame.Position).Magnitude
        if (Range > MaxDistanceSlider.Value) then
            continue
        end

        local Part = Character:FindFirstChild(HitboxDropdown.Value) or RootPart
        local ScreenPos, OnScreen = Camera:WorldToScreenPoint(Part.Position)
        if (not OnScreen) then
            continue
        end

        local ScreenDistance = (Vector2.new(ScreenPos.X, ScreenPos.Y) - ViewportCenter).Magnitude
        if (ScreenDistance > FOVSlider.Value) then
            continue
        end

        if (ScreenDistance < BestDistance) then
            BestDistance = ScreenDistance
            BestPart = Part
        end
    end

    if (not BestPart) then
        return nil
    end

    local Point = BestPart.Position

    -- // Simple prediction
    if (PredictionSlider.Value > 0) then
        local Velocity = BestPart.AssemblyLinearVelocity or Vector3.new()
        local TravelTime = (Point - Camera.CFrame.Position).Magnitude / 1000
        Point = Point + Velocity * TravelTime * PredictionSlider.Value
    end

    return Point
end

    if (not Active) then
        return
    end

    local Target = GetTargetPoint()
    if (not Target) then
        return
    end

    -- // Aim: rotate the camera onto the target
    local Desired = CFrame.lookAt(Camera.CFrame.Position, Target)
    local Smooth = SmoothnessSlider.Value / 100
    if (Smooth >= 1) then
        Camera.CFrame = Desired
    else
        Camera.CFrame = Camera.CFrame:Lerp(Desired, Smooth)
    end
end)
