-- // Helpers
local function GetCharacter(Player)
    return Player.Character
end

local function IsAlive(Character)
    if (not Character) then
        return false
    end

    local Humanoid = Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Character:FindFirstChild("HumanoidRootPart")
    return Humanoid and RootPart and Humanoid.Health > 0
end

-- // Find the best target and return the aim point (or nil)
local function GetTargetPoint()
    local ViewportCenter = Camera.ViewportSize / 2
    local BestPart = nil
    local BestDistance = math.huge

    for _, Player in ipairs(Players:GetPlayers()) do
        if (Player == LocalPlayer) then
            continue
        end

        if (TeamCheckToggle.Value and LocalPlayer.Team and Player.Team and Player.Team == LocalPlayer.Team) then
            continue
        end

        local Character = GetCharacter(Player)
        if (not IsAlive(Character)) then
            continue
        end

        local RootPart = Character:FindFirstChild("HumanoidRootPart")
        if (not RootPart) then
            continue
        end

        local Range = (RootPart.Position - Camera.CFrame.Position).Magnitude
        if (Range > MaxDistanceSlider.Value) then
            continue
        end

        local Part = Character:FindFirstChild(HitboxDropdown.Value) or RootPart
        local ScreenPos, OnScreen = Camera:WorldToScreenPoint(Part.Position)
        if (not OnScreen) then
            continue
        end

        local ScreenDistance = (Vector2.new(ScreenPos.X, ScreenPos.Y) - ViewportCenter).Magnitude
        if (ScreenDistance > FOVSlider.Value) then
            continue
        end

        if (ScreenDistance < BestDistance) then
            BestDistance = ScreenDistance
            BestPart = Part
        end
    end

    if (not BestPart) then
        return nil
    end

    local Point = BestPart.Position

    -- // Simple prediction
    if (PredictionSlider.Value > 0) then
        local Velocity = BestPart.AssemblyLinearVelocity or Vector3.new()
        local TravelTime = (Point - Camera.CFrame.Position).Magnitude / 1000
        Point = Point + Velocity * TravelTime * PredictionSlider.Value
    end

    return Point
end

-- // Main loop - runs after the game's own camera so we win the override

RunService:BindToRenderStep("BlockSpinLegitAim", Enum.RenderPriority.Camera.Value + 1, function()
    local Active = EnabledToggle.Value and EnabledKey:GetState()
    if (not Active) then
        return
    end

    local Target = GetTargetPoint()
    if (not Target) then
        return
    end

    -- // Aim: rotate the camera onto the target
    local Desired = CFrame.lookAt(Camera.CFrame.Position, Target)
    local Smooth = SmoothnessSlider.Value / 100
    if (Smooth >= 1) then
        Camera.CFrame = Desired
    else
        Camera.CFrame = Camera.CFrame:Lerp(Desired, Smooth)
    end
end)

local function EnsureESP(Player)
    local Entry = ESPCache[Player]
    if (not Entry) then
        Entry = {
            Box = Drawing.new("Square"),
            Name = Drawing.new("Text"),
            Health = Drawing.new("Text"),
        }

        Entry.Box.Thickness = 1
        Entry.Box.Transparency = 1
        Entry.Box.Color = Color3.fromRGB(255, 60, 60)

        Entry.Name.Center = true
        Entry.Name.Outline = true
        Entry.Name.OutlineColor = Color3.new(0, 0, 0)
        Entry.Name.Size = 13

        Entry.Health.Center = true
        Entry.Health.Outline = true
        Entry.Health.OutlineColor = Color3.new(0, 0, 0)
        Entry.Health.Size = 12

        ESPCache[Player] = Entry
    end
    return Entry
end

        if (Player == LocalPlayer) then
            local SelfEntry = ESPCache[Player]
            if (SelfEntry) then
                SelfEntry.Box.Visible = false
                SelfEntry.Name.Visible = false
                SelfEntry.Health.Visible = false
            end
            continue
        end

        local Entry = EnsureESP(Player)
        local Character = Player.Character
        local RootPart = Character and Character:FindFirstChild("HumanoidRootPart")
        local Head = Character and Character:FindFirstChild("Head")
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local Show = false

        if (Enabled and RootPart and Head and Humanoid and Humanoid.Health > 0) then
            local IsTeammate = LocalPlayer.Team and Player.Team and Player.Team == LocalPlayer.Team
            if (not (ESPTeamCheckToggle.Value and IsTeammate)) then
                local Range = (RootPart.Position - Camera.CFrame.Position).Magnitude
                if (Range <= MaxDist) then
                    local ScreenHead, OnHead = Camera:WorldToScreenPoint(Head.Position)
                    if (OnHead) then
                        local HeadTop = Head.Position + Head.CFrame.UpVector * (Head.Size.Y / 2)
                        local Feet = RootPart.Position - RootPart.CFrame.UpVector * (RootPart.Size.Y / 2)
                        local ScreenTop, OnTop = Camera:WorldToScreenPoint(HeadTop)
                        local ScreenFeet, OnFeet = Camera:WorldToScreenPoint(Feet)

                        if (OnTop and OnFeet) then
                            local Height = math.abs(ScreenFeet.Y - ScreenTop.Y)
                            local Width = Height * 0.6
                            local TopY = math.min(ScreenTop.Y, ScreenFeet.Y)
                            local BottomY = math.max(ScreenTop.Y, ScreenFeet.Y)
                            local CenterX = (ScreenTop.X + ScreenFeet.X) / 2

                            if (ESPBoxesToggle.Value) then
                                Entry.Box.Visible = true
                                Entry.Box.Color = IsTeammate and Color3.fromRGB(60, 255, 60) or Color3.fromRGB(255, 60, 60)
                                Entry.Box.Position = Vector2.new(CenterX - Width / 2, TopY)
                                Entry.Box.Size = Vector2.new(Width, Height)
                            else
                                Entry.Box.Visible = false
                            end

                            if (ESPNameToggle.Value) then
                                Entry.Name.Visible = true
                                Entry.Name.Text = Player.Name
                                Entry.Name.Position = Vector2.new(CenterX, TopY - 15)
                            else
                                Entry.Name.Visible = false
                            end

                                Entry.Health.Visible = true
                                Entry.Health.Text = string.format("%d/%d", math.floor(Humanoid.Health), math.floor(Humanoid.MaxHealth))
                                Entry.Health.Position = Vector2.new(CenterX, BottomY + 2)
                            else
                                Entry.Health.Visible = false
                            end

                            Show = true
                        end
                    end
                end
            end
        end

        if (not Show) then
            Entry.Box.Visible = false
            Entry.Name.Visible = false
            Entry.Health.Visible = false
        end
    end
end)

task.spawn(function()
    while true do
        pcall(function()
            if HudFrame.Visible then
                local DISPLAYENTRIES = {}
                for _, instance in ipairs(Players:GetChildren()) do
                    if not instance:IsA("Player") then continue end
                    local pObj = instance
                    local pName = pObj.Name
                    local SETITEMS = {}
                    local ITEMTODISPLAY = {}
                    local ITEMCHECK = {}

                    local backpack = pObj:FindFirstChild("Backpack")
                    if backpack then
                        for _, tool in ipairs(backpack:GetChildren()) do
                            if tool:IsA("Tool") then table.insert(ITEMCHECK, tool) end
                        end
                    end

                    local character = pObj.Character
                    if character then
                        for _, child in ipairs(character:GetChildren()) do
                            if child:IsA("Tool") then table.insert(ITEMCHECK, child) end
                        end
                    end

                    for _, tool in ipairs(ITEMCHECK) do
                        local identified = false
                        for _, part in ipairs(tool:GetChildren()) do
                            local partName = part.Name
                            local prefix = "Meshes/"
                            if string.sub(partName, 1, #prefix) == prefix then
                                local subName = string.lower(string.sub(partName, #prefix + 1))
                                for _, entry in ipairs(ITEM_DATABASE) do
                                    local pattern = string.lower(entry.meshPattern)
                                    local matched = false
                                    if entry.isPrefix then
                                        if string.sub(subName, 1, #pattern) == pattern then matched = true end
                                    else
                                        if subName == pattern then matched = true end
                                    end

                                    if matched then
                                        if not SETITEMS[entry.displayName] then
                                            SETITEMS[entry.displayName] = true
                                            table.insert(ITEMTODISPLAY, "- " .. entry.displayName)
                                        end
                                        identified = true
                                        break
                                    end
                                end
                            elseif partName == "Butt" then
                                if not SETITEMS["Axe"] then
                                    SETITEMS["Axe"] = true
                                    table.insert(ITEMTODISPLAY, "- Axe")
                                end
                                identified = true
                            elseif partName == "Bat" then
                                if not SETITEMS["Bat"] then
                                    SETITEMS["Bat"] = true
                                    table.insert(ITEMTODISPLAY, "- Bat")
                                end
                                identified = true
                            end
                            if identified then break end
                        end
                    end

                    if #ITEMTODISPLAY > 0 then
                        table.insert(DISPLAYENTRIES, pName .. ":\n" .. table.concat(ITEMTODISPLAY, "\n"))
                    end
                end

                local resultStr = table.concat(DISPLAYENTRIES, "\n\n")
                if resultStr == "" then
                    HudTextLabel.Text = "No player inventories detected."
                else
                    HudTextLabel.Text = resultStr
                end
            end
        end)
        task.wait(0.5)
    end
end)

    local out = {}
    for _, file in ipairs(listfiles(CONFIG_FOLDER)) do
        local Name = file:match("([^/\\]+)%.json$")
        if (Name) then
            table.insert(out, Name)
        end
    end
    return out
end

local function RefreshConfigList()
    ConfigListDropdown.Values = GetConfigList()
    ConfigListDropdown:SetValues()
    ConfigListDropdown:SetValue(nil)
end

local function SaveConfig(Name)
    if (not Name or Name:gsub("%s", "") == "") then
        return Library:Notify("Invalid config name (empty)", 2)
    end

    local Data = { objects = {} }

    for Idx, Toggle in next, SaveToggles do
        table.insert(Data.objects, { type = "Toggle", idx = Idx, value = Toggle.Value })
    end

    for Idx, Option in next, SaveOptions do
        local Type = Option.Type
        if (Type == "Slider") then
            table.insert(Data.objects, { type = "Slider", idx = Idx, value = tostring(Option.Value) })
        elseif (Type == "Dropdown") then
            table.insert(Data.objects, { type = "Dropdown", idx = Idx, value = Option.Value, mutli = Option.Multi })
        elseif (Type == "KeyPicker") then
            table.insert(Data.objects, { type = "KeyPicker", idx = Idx, mode = Option.Mode, key = Option.Value })
        end
    end

    local Success, Encoded = pcall(HttpService.JSONEncode, HttpService, Data)
    if (not Success) then
        return Library:Notify("Failed to save config: encode error", 3)
    end

    writefile(CONFIG_FOLDER .. "/" .. Name .. ".json", Encoded)
    Library:Notify(string.format('Created config "%s"', Name), 3)
    RefreshConfigList()
end

local function LoadConfig(Name)
    if (not Name) then
        return Library:Notify("No config selected", 2)
    end

    local Path = CONFIG_FOLDER .. "/" .. Name .. ".json"
    if (not isfile(Path)) then
        return Library:Notify("Config not found", 2)
    end

    local Success, Decoded = pcall(HttpService.JSONDecode, HttpService, readfile(Path))
    if (not Success) then
        return Library:Notify("Failed to load config: decode error", 3)
    end

    for _, Object in next, Decoded.objects do
        local Toggle = SaveToggles[Object.idx]
        local Option = SaveOptions[Object.idx]

        if (Toggle and Object.type == "Toggle") then
            Toggle:SetValue(Object.value)
        elseif (Option) then
            if (Object.type == "Slider") then
                Option:SetValue(Object.value)
            elseif (Object.type == "Dropdown") then
                Option:SetValue(Object.value)
            elseif (Object.type == "KeyPicker") then
                Option:SetValue({ Object.key, Object.mode })
            end
        end
    end

    Library:Notify(string.format('Loaded config "%s"', Name), 3)
end
